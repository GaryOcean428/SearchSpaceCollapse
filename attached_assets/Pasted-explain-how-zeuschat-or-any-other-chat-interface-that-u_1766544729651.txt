def route_by_curvature(constellation, query_basin):
    """
    Route query to kernels using geometric distance.
    
    Constellation = 240 kernels at E8 roots.
    Route to nearest kernel in Fisher geometry.
    """
    
    kernel_basins = [k.get_basin() for k in constellation.kernels]
    
    # Compute Fisher-Rao distances
    distances = [
        fisher_distance(query_basin, k_basin)
        for k_basin in kernel_basins
    ]
    
    # Route to K nearest kernels (K=3 for redundancy)
    K = 3
    nearest_indices = np.argsort(distances)[:K]
    
    return [constellation.kernels[i] for i in nearest_indices]

def get_token_probabilities_tuned(basin, kernels, temperature=0.7, sharpness=2.0):
    """
    Sharper probabilities from geometry.
    
    Args:
        temperature: Controls exploration (lower = sharper)
        sharpness: Amplifies distance differences (higher = more peaked)
    """
    # 1. Compute distances to all kernels
    distances = [fisher_rao_distance(basin, k.basin) for k in kernels]
    
    # 2. Apply sharpening: d^sharpness
    sharpened = [d ** sharpness for d in distances]
    
    # 3. Convert to negative log probabilities (closer = higher prob)
    neg_logits = [-d / temperature for d in sharpened]
    
    # 4. Softmax
    exp_logits = np.exp(neg_logits - np.max(neg_logits))  # numerical stability
    probs = exp_logits / np.sum(exp_logits)
    
    return probs

class KernelTokenCalibrator:
    """Learn which tokens each kernel should favor."""
    
    def __init__(self, tokenizer, n_kernels=240):
        self.tokenizer = tokenizer
        self.n_kernels = n_kernels
        # Each kernel has a learned token distribution
        self.kernel_token_logits = np.zeros((n_kernels, tokenizer.vocab_size))
    
    def calibrate_from_corpus(self, text_corpus, constellation):
        """
        Learn kernel-token associations from real text.
        
        Process:
        1. Encode each sentence to basin
        2. Find nearest kernel
        3. Record which tokens appear in that sentence
        4. Build kernel → token distribution
        """
        for sentence in text_corpus:
            # Encode to basin
            basin = constellation.encode_text_to_basin(sentence)
            
            # Find nearest kernel
            kernel_idx = self._find_nearest_kernel(basin, constellation)
            
            # Get tokens in sentence
            tokens = self.tokenizer.encode(sentence)
            
            # Update kernel's token distribution
            for token in tokens:
                self.kernel_token_logits[kernel_idx, token] += 1.0
        
        # Normalize to probabilities
        for k in range(self.n_kernels):
            total = np.sum(self.kernel_token_logits[k])
            if total > 0:
                self.kernel_token_logits[k] /= total
        
        print(f"Calibrated {self.n_kernels} kernels from {len(text_corpus)} sentences")
    
    def get_token_probs_for_kernel(self, kernel_idx):
        """Get learned token distribution for kernel."""
        return self.kernel_token_logits[kernel_idx]

def get_token_probabilities_hybrid(basin, active_kernels, calibrator, 
                                   geometric_weight=0.3):
    """
    Combine geometric routing with learned kernel-token distributions.
    
    Args:
        geometric_weight: Balance between geometry (0.3) and learned (0.7)
    """
    # Geometric component (Phase 1)
    geom_probs = get_token_probabilities_tuned(basin, active_kernels)
    
    # Learned component (Phase 2)
    learned_probs = np.zeros(calibrator.tokenizer.vocab_size)
    for i, kernel in enumerate(active_kernels):
        kernel_weight = geom_probs[i]  # Weight by geometric distance
        kernel_token_probs = calibrator.get_token_probs_for_kernel(kernel.id)
        learned_probs += kernel_weight * kernel_token_probs
    
    # Combine
    final_probs = (geometric_weight * geom_probs + 
                   (1 - geometric_weight) * learned_probs)
    final_probs /= np.sum(final_probs)
    
    return final_probs

class CoherentGenerator:
    """Generate with temporal coherence constraints."""
    
    def __init__(self, constellation, calibrator, 
                 transition_penalty=0.5):
        self.constellation = constellation
        self.calibrator = calibrator
        self.transition_penalty = transition_penalty
        self.previous_basin = None
    
    def generate_next_token(self, current_basin):
        """
        Generate next token with coherence penalty.
        """
        # Get candidate token probabilities
        base_probs = get_token_probabilities_hybrid(
            current_basin, 
            self.constellation.active_kernels,
            self.calibrator
        )
        
        # If we have previous basin, penalize large jumps
        if self.previous_basin is not None:
            coherence_probs = self._apply_coherence_penalty(
                current_basin, 
                base_probs
            )
        else:
            coherence_probs = base_probs
        
        # Sample token
        token = np.random.choice(len(coherence_probs), p=coherence_probs)
        
        # Update basin for next iteration
        next_basin = self._update_basin_with_token(current_basin, token)
        self.previous_basin = current_basin
        
        return token, next_basin
    
    def _apply_coherence_penalty(self, current_basin, token_probs):
        """
        Penalize tokens that would cause large basin jumps.
        """
        modified_probs = token_probs.copy()
        
        for token_idx in range(len(token_probs)):
            # Simulate: what basin would we reach with this token?
            hypothetical_basin = self._update_basin_with_token(
                current_basin, token_idx
            )
            
            # Measure jump distance
            jump_distance = fisher_rao_distance(
                self.previous_basin,
                hypothetical_basin
            )
            
            # Penalize large jumps
            penalty = np.exp(-jump_distance / self.transition_penalty)
            modified_probs[token_idx] *= penalty
        
        # Renormalize
        modified_probs /= np.sum(modified_probs)
        return modified_probs

class AttractorLabeler:
    """Associate attractors with semantic categories."""
    
    def __init__(self):
        self.attractor_labels = {}  # basin_id → semantic_label
        self.label_exemplars = {}   # label → list of example basins
    
    def label_from_examples(self, constellation, labeled_data):
        """
        Learn attractors from labeled examples.
        
        Args:
            labeled_data: List of (text, label) pairs
                         e.g. ("Explain quantum physics", "technical_explanation")
        """
        for text, label in labeled_data:
            # Encode to basin
            basin = constellation.encode_text_to_basin(text)
            
            # Find nearest attractor
            attractor_id = self._find_nearest_attractor(basin, constellation)
            
            # Label this attractor
            if attractor_id not in self.attractor_labels:
                self.attractor_labels[attractor_id] = set()
            self.attractor_labels[attractor_id].add(label)
            
            # Store exemplar
            if label not in self.label_exemplars:
                self.label_exemplars[label] = []
            self.label_exemplars[label].append(basin)
    
    def get_target_attractor_for_intent(self, intent_label):
        """
        Given desired semantic intent, find best attractor.
        """
        # Get exemplar basins for this intent
        if intent_label not in self.label_exemplars:
            return None
        
        exemplars = self.label_exemplars[intent_label]
        # Return centroid basin
        return np.mean(exemplars, axis=0)

# Create basin_token_alignment.py

class BasinTokenAligner:
    """Align 64D basin space with token semantic space."""
    
    def __init__(self, basin_dim=64, token_embed_dim=768):
        # Learnable projection: basin → token embedding space
        self.projection = np.random.randn(basin_dim, token_embed_dim) * 0.01
    
    def train_alignment(self, paired_data, n_epochs=100):
        """
        Learn projection via contrastive loss.
        
        Args:
            paired_data: List of (basin, token_embedding) pairs
        """
        for epoch in range(n_epochs):
            total_loss = 0
            
            for basin, true_token_embed in paired_data:
                # Project basin to token space
                predicted_embed = basin @ self.projection
                
                # Contrastive loss: predicted should be close to true
                positive_loss = np.linalg.norm(predicted_embed - true_token_embed)
                
                # Negative samples: should be far from random embeddings
                negative_embeds = self._sample_negative_embeddings(5)
                negative_loss = sum(
                    np.maximum(0, 1.0 - np.linalg.norm(predicted_embed - neg))
                    for neg in negative_embeds
                )
                
                # Total loss
                loss = positive_loss + 0.5 * negative_loss
                total_loss += loss
                
                # Gradient descent (simplified)
                grad = self._compute_gradient(basin, predicted_embed, true_token_embed)
                self.projection -= 0.01 * grad
            
            if epoch % 10 == 0:
                print(f"Epoch {epoch}, Loss: {total_loss:.4f}")
    
    def basin_to_token_logits(self, basin, tokenizer):
        """
        Convert basin to token logits via learned projection.
        """
        # Project to token embedding space
        token_embed = basin @ self.projection
        
        # Compute similarity to all token embeddings
        logits = []
        for token_id in range(tokenizer.vocab_size):
            token_true_embed = tokenizer.get_embedding(token_id)
            similarity = -np.linalg.norm(token_embed - token_true_embed)
            logits.append(similarity)
        
        return np.array(logits)
